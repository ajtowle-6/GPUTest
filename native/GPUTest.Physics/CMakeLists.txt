cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
project(GPUTestPhysics LANGUAGES C CXX)
option(GPUTEST_PROFILE_ZONES "Enable PhysX host profiling zones" OFF)
if(GPUTEST_PROFILE_ZONES)
	add_compile_definitions(PX_PROFILE=1)
endif()
set(PHYSX_ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.deps/physx-5.4.2/physx" CACHE PATH "Pinned PhysX SDK")
set(TARGET_BUILD_PLATFORM windows CACHE STRING "PhysX platform")
set(PX_OUTPUT_ARCH x86 CACHE STRING "PhysX architecture")
set(PX_OUTPUT_LIB_DIR "${CMAKE_BINARY_DIR}/sdk" CACHE PATH "PhysX libraries")
set(PX_OUTPUT_BIN_DIR "${CMAKE_BINARY_DIR}/sdk" CACHE PATH "PhysX binaries")
set(PX_GENERATE_STATIC_LIBRARIES ON CACHE BOOL "Static SDK")
set(NV_USE_STATIC_WINCRT ON CACHE BOOL "Static CRT")
set(NV_USE_DEBUG_WINCRT OFF CACHE BOOL "Release CRT")
set(PX_BUILDSNIPPETS OFF CACHE BOOL "No SDK samples")
set(PX_BUILDPVDRUNTIME OFF CACHE BOOL "No PVD runtime")
add_subdirectory("${PHYSX_ROOT_DIR}/compiler/public" physx)
add_library(GPUTest.Physics SHARED PhysicsBridge.cpp)
target_include_directories(GPUTest.Physics PRIVATE "${PHYSX_ROOT_DIR}/include")
target_compile_features(GPUTest.Physics PRIVATE cxx_std_17)
target_compile_definitions(GPUTest.Physics PRIVATE PX_PHYSX_STATIC_LIB NDEBUG _ITERATOR_DEBUG_LEVEL=0)
set_property(TARGET GPUTest.Physics PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_link_libraries(GPUTest.Physics PRIVATE PhysX PhysXCommon PhysXFoundation PhysXExtensions PhysXPvdSDK)
set_target_properties(GPUTest.Physics PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime/$<CONFIG>")
